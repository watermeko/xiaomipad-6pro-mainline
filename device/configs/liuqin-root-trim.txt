# Root filesystem trim for the liuqin native root (Ubuntu 26.04 + Plasma Mobile).
#
# Read by the `debs` stage of tools/build-liuqin-native-root.sh after the liuqin
# .deb set is installed, inside the same ARM64 chroot.  The stage purges every
# `[purge]` package, autoremoves what that orphaned, and then fails the build
# unless every `[keep]` package is still installed and every `[purge]` package
# (plus its D-Bus names) is really gone.  A trim that silently costs the
# desktop therefore breaks the build instead of the panel.
#
# Everything here is measured on the 8 GiB unit: see docs/MEMORY.md.
#
# `[purge]` entries are package names only.  `[keep]` entries are the runtime
# the desktop, the sensors, the audio path and the fingerprint login need; they
# exist so that a dependency change in a future distro snapshot fails here.
#
# The desktop was GNOME/Phosh until the Plasma Mobile migration.  Every GNOME
# session package is now in [purge], including gdm3: its greeter *is*
# gnome-shell, so keeping the login manager would have kept the whole Shell
# stack for a screen that autologin skips.  sddm starts the same
# plasma-mobile.desktop session instead.

[mask-user]
# Per-account tweaks the product applies to whichever account logs in.  These
# are written into the live user's own configuration, not into /etc: they
# belong to a session, and another account on the same tablet may legitimately
# want the opposite.
#
#   unit:<unit>          mask a systemd --user unit
#   autostart:<file>     hide an XDG autostart entry in ~/.config/autostart
#
# Applied by /usr/lib/liuqin/apply-root-tweaks.sh --user-only, which
# liuqin-session-tweaks.service runs once per login.
#
# baloo_file is Plasma's file indexer.  It is started from
# /etc/xdg/autostart/baloo_file.desktop on every login and this tablet has no
# use for a full-text index of its home directory: the measured session on the
# 8 GiB unit ran no baloo process at all (the indexer exits when there is no
# indexable content), so hiding the entry costs nothing today and keeps a
# first-boot index from appearing once /home has files.
autostart:baloo_file.desktop
